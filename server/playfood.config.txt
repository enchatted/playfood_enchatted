# [live] Turn statistics recording and the /panel page on or off.
# no  = nothing is recorded, and /panel plus its data API are all disabled.
PANEL_ENABLED    = no

# [rebuild] Secret phrase that accompanies every statistics submission.
# Put your own random phrase here. The server picks up a change within 30s,
# BUT the game only sends the phrase it was built with — so after changing
# this you must rebuild the game with the matching VITE_PANEL_SECRET, or
# every event is silently rejected. See the Technical Documentation (Deliverable 5).
PANEL_SECRET     = change-this-to-your-own-phrase

# [live] How many months of data we keep. Older records are deleted on the
# next daily cleanup (runs every 24h and at startup).
# New installations: use the supplied 24-month setting.
# Existing installations: update this key only; preserve all other site settings.
# Separate backups and CSV exports need their own retention management.

# [restart] Time zone — decides what "today" means in the date filters.
# Applied once at startup; restart the server after changing it.
TIMEZONE         = Europe/Athens

# [restart] Server port. Bound once at startup; restart to change.
PORT             = 4000

# [live] Safety valve: maximum records accepted per day.
MAX_EVENTS_PER_DAY = 200000
